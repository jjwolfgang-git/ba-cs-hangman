# The callback when the Guess button is pressed: 
def letterGuessed():

    global num_guesses, incorrect, correct

    # Get the letter from the GUI widget and immediately convert it to uppercase
    letter = guess_entry.get().upper()
    
    # Make sure a valid letter was provided
    # Make sure the letter was not already guessed
    if not isGuessValid(letter, incorrect, correct):
        messagebox.showwarning("Invalid Input!", "Please enter a letter that hasn't been guessed")
        return
    
    letterAppears = analyzeGuess(letter, wordStruct)
    
    if letterAppears == 0: #letter does not appear in the word
        incorrect.append(letter)
        num_guesses = num_guesses - 1
    else:
        correct.append(letter)
        
    if num_guesses == 0: #player loses
        messagebox.showerror("Game Over", f"The word was {word}. Please play again!")

    elif isWordFound(wordStruct) == True: #player wins
        messagebox.showinfo("You WIN!", f"You had {num_guesses} incorrect guesses remaining")
        
    display(wordStruct, num_guesses, incorrect, word_label, incorrect_label, attempts_label, guess_entry)
